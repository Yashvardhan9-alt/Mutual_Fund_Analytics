-- ==========================================
-- Mutual Fund Analytics
-- Analytical SQL Queries
-- ==========================================

----------------------------------------------------
-- Query 1 : Total NAV Records
----------------------------------------------------

SELECT COUNT(*) AS total_nav_records
FROM fact_nav;

----------------------------------------------------
-- Query 2 : Total Investment Amount
----------------------------------------------------

SELECT SUM(amount_inr) AS total_investment
FROM fact_transactions;

----------------------------------------------------
-- Query 3 : Transaction Type Distribution
----------------------------------------------------

SELECT
    transaction_type,
    COUNT(*) AS total_transactions
FROM fact_transactions
GROUP BY transaction_type
ORDER BY total_transactions DESC;

----------------------------------------------------
-- Query 4 : Top 10 Funds by Investment
----------------------------------------------------

SELECT
    amfi_code,
    SUM(amount_inr) AS total_amount
FROM fact_transactions
GROUP BY amfi_code
ORDER BY total_amount DESC
LIMIT 10;

----------------------------------------------------
-- Query 5 : KYC Status Distribution
----------------------------------------------------

SELECT
    kyc_status,
    COUNT(*) AS investors
FROM fact_transactions
GROUP BY kyc_status;

----------------------------------------------------
-- Query 6 : State-wise Investment
----------------------------------------------------

SELECT
    state,
    SUM(amount_inr) AS total_investment
FROM fact_transactions
GROUP BY state
ORDER BY total_investment DESC;

----------------------------------------------------
-- Query 7 : Average NAV
----------------------------------------------------

SELECT
    amfi_code,
    AVG(nav) AS average_nav
FROM fact_nav
GROUP BY amfi_code;

----------------------------------------------------
-- Query 8 : Best Performing Funds
----------------------------------------------------

SELECT
    amfi_code,
    return_1yr_pct
FROM fact_performance
ORDER BY return_1yr_pct DESC
LIMIT 10;

----------------------------------------------------
-- Query 9 : Highest Expense Ratio
----------------------------------------------------

SELECT
    amfi_code,
    expense_ratio_pct
FROM fact_performance
ORDER BY expense_ratio_pct DESC
LIMIT 10;

----------------------------------------------------
-- Query 10 : Average Transaction Amount
----------------------------------------------------

SELECT
    transaction_type,
    AVG(amount_inr) AS average_amount
FROM fact_transactions
GROUP BY transaction_type;
